/* ---------- PARTICLES ---------- */

(function(){

  const canvas = document.getElementById('particle-canvas');

  const ctx = canvas.getContext('2d');

  let particles = [], w, h;

  function resize(){ w = canvas.width = window.innerWidth; h = canvas.height = window.innerHeight; }

  function init(){

    particles = [];

    const count = Math.min(80, Math.floor(w * h / 18000));

    for(let i=0;i<count;i++){

      particles.push({

        x: Math.random()*w, y: Math.random()*h,

        vx: (Math.random()-0.5)*0.3, vy: (Math.random()-0.5)*0.3,

        r: Math.random()*1.8+0.4, a: Math.random()*0.5+0.1

      });

    }

  }

  function draw(){

    ctx.clearRect(0,0,w,h);

    for(const p of particles){

      p.x += p.vx; p.y += p.vy;

      if(p.x<0)p.x=w; if(p.x>w)p.x=0; if(p.y<0)p.y=h; if(p.y>h)p.y=0;

      ctx.beginPath();

      ctx.arc(p.x, p.y, p.r, 0, Math.PI*2);

      ctx.fillStyle = 'rgba(162,89,255,'+p.a+')';

      ctx.fill();

    }

    requestAnimationFrame(draw);

  }

  resize(); init(); draw();

  window.addEventListener('resize', ()=>{ resize(); init(); });

})();

(function(){

  const phrases = ["schedule a meeting...","book an appointment...","qualify a new lead...","take a message...","handle common FAQs..."];

  const el = document.getElementById('tw-text');

  let pIndex=0, charIndex=0, isDeleting=false;

  function type(){

    const current = phrases[pIndex];

    if(isDeleting){ el.textContent = current.substring(0, charIndex-1); charIndex--; }

    else { el.textContent = current.substring(0, charIndex+1); charIndex++; }

    let speed = isDeleting ? 50 : 100;

    if(!isDeleting && charIndex===current.length){ speed=2000; isDeleting=true; }

    else if(isDeleting && charIndex===0){ isDeleting=false; pIndex=(pIndex+1)%phrases.length; speed=500; }

    setTimeout(type, speed);

  }

  type();

})();

/* ---------- CURRENCY CONVERTER ---------- */

let rates = null;

async function loadRates(){

  try{

    const res = await fetch('https://open.er-api.com/v6/latest/USD');

    const data = await res.json();

    rates = data.rates;

    const updated = new Date(data.time_last_update_unix * 1000).toLocaleString();

    document.getElementById('conv-updated').textContent = 'Rates updated: ' + updated;

    convert();

  }catch(err){

    document.getElementById('conv-updated').textContent = 'Live rates unavailable';

  }

}

function convert(){

  if(!rates) return;

  const usd = parseFloat(document.getElementById('usd-input').value) || 0;

  document.getElementById('gbp-val').textContent = (usd * rates.GBP).toFixed(2);

  document.getElementById('aud-val').textContent = (usd * rates.AUD).toFixed(2);

}

document.getElementById('usd-input').addEventListener('input', convert);

loadRates();
